cmake_minimum_required(VERSION 3.20)
project(mixed_atmospheres_liquid_reference LANGUAGES CXX)

option(MA_REFERENCE_SANITIZERS "Enable address/undefined sanitizers for the isolated reference" OFF)
if(MA_REFERENCE_SANITIZERS AND (MSVC OR NOT CMAKE_CXX_COMPILER_ID MATCHES "^(GNU|Clang)$"))
  message(FATAL_ERROR "MA_REFERENCE_SANITIZERS requires GCC or Clang with address/undefined sanitizer support; MSVC is unsupported")
endif()

# Independent thermodynamic test project. Root CMake separately links the
# implementation for its explicit experimental FluidWorld geometry mode.
add_library(liquid_thermodynamic_reference STATIC liquid_reference.cpp)
target_compile_features(liquid_thermodynamic_reference PUBLIC cxx_std_20)
target_include_directories(liquid_thermodynamic_reference PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(liquid_reference_tests liquid_reference_tests.cpp)
target_link_libraries(liquid_reference_tests PRIVATE liquid_thermodynamic_reference)
foreach(target liquid_thermodynamic_reference liquid_reference_tests)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  if(MA_REFERENCE_SANITIZERS)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-sanitize-recover=all)
  endif()
endforeach()
if(MA_REFERENCE_SANITIZERS)
  target_link_options(liquid_reference_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
endif()
enable_testing()
add_test(NAME liquid_reference_thermodynamics COMMAND liquid_reference_tests)
if(MA_REFERENCE_SANITIZERS)
  set_tests_properties(liquid_reference_thermodynamics PROPERTIES
    ENVIRONMENT "ASAN_OPTIONS=detect_leaks=1:halt_on_error=1;UBSAN_OPTIONS=halt_on_error=1:print_stacktrace=1")
endif()
